@ExtendWith(MockitoExtension.class)

    @Mock

    @Test
    public void testDecodeAs() throws Exception {
        Map<String, String> supersetMap = new HashMap<>();
        supersetMap.put("UTF-8", "UTF-16");
        supersetMap.put("ISO-8859-1", "UTF-8");
        supersetMap.put("ASCII", "UTF-8");
        CharsetSupersets.SUPERSET_MAP = Collections.unmodifiableMap(supersetMap);
        CharsetSupersets charsetSupersets = CharsetSupersets.class.newInstance();
        Method decodeAsMethod = CharsetSupersets.class.getDeclaredMethod("decodeAs", Charset.class);
        decodeAsMethod.setAccessible(true);
        // Test with null input
        assertNull(decodeAsMethod.invoke(charsetSupersets, (Charset) null));
        // Test with UTF-8 input
        assertEquals("UTF-16", decodeAsMethod.invoke(charsetSupersets, detected).toString());
        verify(detected).name();
        // Test with ISO-8859-1 input
        assertEquals("UTF-8", decodeAsMethod.invoke(charsetSupersets, detected).toString());
        verify(detected).name();
        // Test with ASCII input
        assertEquals("UTF-8", decodeAsMethod.invoke(charsetSupersets, detected).toString());
        verify(detected).name();
        // Test with unknown input
        assertNull(decodeAsMethod.invoke(charsetSupersets, Charset.forName("UNKNOWN")));
    }
}
